Frontend URL resolution (already implemented — do not change)

ts// src/lib/apiFetch.ts
const API_BASE = import.meta.env.VITE_API_BASE_URL ?? ''; // = "https://api.thejourney-ma.org" in prod

export function resolveStorageUrl(url: string | null | undefined): string | null {
  if (!url) return null;
  if (url.startsWith('http://') || url.startsWith('https://')) return url;
  if (url.startsWith('/storage/') || url.startsWith('/uploads/')) {
    return API_BASE ? `${API_BASE}${url}` : url;
  }
  // bare UUID filename → treat as /storage/media/<uuid>.<ext>
  if (/^[0-9a-f-]{36}\.[a-z0-9]+$/i.test(url)) {
    const path = `/storage/media/${url}`;
    return API_BASE ? `${API_BASE}${path}` : path;
  }
  return url;
}


The canonical upload helper (backend)

Create or reuse this private method in any controller that needs it:

phpuse App\Models\MediaAsset;
use Illuminate\Http\UploadedFile;
use Illuminate\Support\Facades\Storage;
use Illuminate\Support\Str;

private function storeMediaAsset(UploadedFile $file, $userId = 'system', string $alt = ''): MediaAsset
{
    $origName = $file->getClientOriginalName();
    $mime     = $file->getMimeType() ?? 'application/octet-stream';
    $ext      = $file->getClientOriginalExtension() ?: 'bin';
    $filename = Str::uuid() . '.' . $ext;

    Storage::disk('public')->put('media/' . $filename, file_get_contents($file->getRealPath()));

    return MediaAsset::create([
        'file_name'     => $origName,
        'file_type'     => $mime,
        'file_size'     => $file->getSize(),
        'file_url'      => '/storage/media/' . $filename,
        'thumbnail_url' => '/storage/media/' . $filename,
        'alt_text'      => $alt ?: pathinfo($origName, PATHINFO_FILENAME),
        'uploaded_by'   => $userId,
    ]);
}
